\end{itemize}

\begin{frame}
\frametitle{Chỉ số chất lượng: sáu chỉ báo, ba chiều trọng số bằng nhau}
\begin{dinhnghia}[Chỉ số chất lượng thị trường]
Với sàn $j$ và tháng $t$, mỗi chiều là trung bình điểm z của các chỉ báo thành phần:
\[
\MQI_{jt} = \frac{\vn{TK}_{jt} + \vn{HQ}_{jt} + \vn{MB}_{jt}}{3}
\]
\end{dinhnghia}
\begin{itemize}
\item Thanh khoản (TK): Amihud, tỷ lệ ngày lợi suất 0, Corwin--Schultz
\item Hiệu quả thông tin (HQ): độ lệch tỷ số phương sai khỏi 1, độ trễ giá
\item Minh bạch (MB): số ngày trễ hạn công bố báo cáo quý
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Khả năng đầu tư: rổ chuyển sàn kém rổ HOSE bao nhiêu}
\begin{dinhnghia}[Khoảng cách khả năng đầu tư]
Lợi suất tương đương chắc chắn của mô hình $m$: $U_m = \mu - \rho\sigma^2/2$; trung bình trên 7 mô hình:
\[
\bar{G} = \frac{1}{7}\sum_{m=1}^{7} \bigl[U_m(\vn{H}) - U_m(\vn{CS})\bigr]
\]
\end{dinhnghia}
\begin{itemize}
\item Rổ CS: tối đa 30 mã chuyển sàn; rổ H: mã HOSE ghép cặp
\item $\mu$, $\sigma^2$: trung bình, phương sai lợi suất vượt trội sau chi phí; $\rho = 5$
\item Kỳ vọng: $\bar{G}$ thu hẹp sau chuyển sàn; 20 cửa sổ trước, 8 sau: \al{chỉ ước lượng}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Thị trường cổ phiếu và điểm GFCI: liên hệ có điều kiện}
\begin{dinhnghia}[Mô hình cho điểm GFCI]
Điểm GFCI chuẩn hóa $Z_{it}$ của trung tâm $i$ ở kỳ $t$:
\[
Z_{it} = \alpha_i + \tau_t + \beta \SMD_{it} + \lambda W_{it} + \varepsilon_{it}
\]
\end{dinhnghia}
\begin{itemize}
\item $\SMD$: vòng quay, vốn hóa / GDP; $W$: thu nhập, quản trị nhà nước
\item $\alpha_i$, $\tau_t$: hiệu ứng cố định trung tâm, kỳ; cụm theo quốc gia; kỳ vọng $\beta > 0$
\item GFCI có nhân tố giao dịch cổ phiếu: $\beta$ \al{không là tác động nhân quả}
\end{itemize}
\end{frame}

\frametitle{Phụ lục: \HH{4} chỉ được ước lượng, kèm khoảng tin cậy bootstrap}
\begin{itemize}
\item $\mu$, $\sigma^2$: lợi suất vượt trội tuần sau chi phí, nhân 52; $\rho = 5$, độ nhạy ở 3 và 10
\item Chi phí A: tỷ lệ cố định nhân vòng quay; chi phí B: phí, thuế, nửa chênh lệch mua-bán
\item $\bar{G} > 0$: rổ CS có lợi suất tương đương chắc chắn thấp hơn rổ H
\item Khoảng tin cậy: bootstrap khối dừng trên lợi suất tuần của hai rổ, lấy mẫu đồng thời
\item Cửa sổ sau còn chứa dữ liệu trước: \MM{1} là phép so sánh sạch nhất
\item Tỷ số Sharpe là thước đo phụ, kiểm định theo Ledoit và Wolf (2008)
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Phụ lục: nghiên cứu sự kiện kiểm tra xu hướng song song}
\[
A_{it} = \alpha_i + \tau_t + \sum_k \delta_k D_i T_{kt} + \varepsilon_{it}
\]
\begin{itemize}
\item $T_{kt} = 1$ nếu tháng $t$ là $k$; $k$ từ 1/2024 đến 12/2028; gốc 6/2025; trước 1/2024 gộp
\item Kiểm định đồng thời $\delta_k = 0$ từ 1/2024 đến 5/2025
\item Độ nhạy trước vi phạm xu hướng song song: Rambachan và Roth (2023)
\item 18 tháng chuyển tiếp giữa tháng gốc và giai đoạn sau: khoảng tin cậy có thể rộng
\end{itemize}
\end{frame}
