\begin{frame}
\frametitle{Chỉ số chất lượng thị trường: sáu chỉ báo, ba chiều cùng trọng số}
\begin{dinhnghia}[Chỉ số chất lượng thị trường]
Với sàn $j$ và tháng $t$, mỗi chiều là trung bình điểm z của các chỉ báo thành phần:
\[
\MQI_{jt} = \frac{\vn{TK}_{jt} + \vn{HQ}_{jt} + \vn{MB}_{jt}}{3}
\]
\end{dinhnghia}
\begin{itemize}
\item Thanh khoản (TK): Amihud, tỷ lệ ngày lợi suất 0, chênh lệch mua-bán Corwin--Schultz
\item Hiệu quả thông tin (HQ): độ lệch tỷ số phương sai khỏi 1, độ trễ giá
\item Minh bạch (MB): số ngày trễ hạn công bố báo cáo quý
\end{itemize}
\end{frame}

Nếu không chuyển sàn, chênh lệch log Amihud giữa hai nhóm không đổi theo thời gian:
\[
A_{it} = \alpha_i + \tau_t + \delta D_i\Post_t + \varepsilon_{it}, \quad E[\varepsilon_{it} \mid D_i, \Post_t, \alpha_i, \tau_t] = 0

\item $A_{it}$: log Amihud; $D_i = 1$: mã chuyển sàn; $\Post_t = 1$ từ 1/2027

\item Đối chứng: mã HOSE ghép theo vốn hóa, ngành, Amihud tại 30/6/2025

\begin{frame}
\frametitle{Khả năng đầu tư: rổ chuyển sàn kém rổ HOSE bao nhiêu}
\begin{dinhnghia}[Khoảng cách khả năng đầu tư]
Lợi suất tương đương chắc chắn của mô hình $m$: $U_m = \mu - \rho\sigma^2/2$; trung bình trên 7 mô hình:
\[
\bar{G} = \frac{1}{7}\sum_{m=1}^{7} \bigl[U_m(\vn{H}) - U_m(\vn{CS})\bigr]
\]
\end{dinhnghia}
\begin{itemize}
\item Đo nhà đầu tư đa dạng hóa thu được bao nhiêu sau chi phí giao dịch, có tính rủi ro
\item Rổ CS: tối đa 30 mã chuyển sàn; rổ H: mã HOSE ghép cặp; $\rho = 5$
\item Kỳ vọng: $\bar{G}$ thu hẹp sau chuyển sàn; 20 cửa sổ trước, 8 sau: \al{chỉ ước lượng}
\end{itemize}
\end{frame}

\begin{frame}
\frametitle{Thị trường cổ phiếu và điểm GFCI: liên hệ có điều kiện}
\begin{dinhnghia}[Mô hình cho điểm GFCI]
Điểm GFCI chuẩn hóa $Z_{it}$ của trung tâm $i$ ở kỳ $t$:
\[
Z_{it} = \alpha_i + \tau_t + \beta \SMD_{it} + \lambda W_{it} + \varepsilon_{it}
\]
\end{dinhnghia}
\begin{itemize}
\item $\SMD$: vòng quay, vốn hóa / GDP; $W$: thu nhập, quản trị nhà nước
\item $\alpha_i$, $\tau_t$: hiệu ứng cố định trung tâm, kỳ; cụm theo quốc gia; kỳ vọng $\beta > 0$
\item GFCI có nhân tố giao dịch cổ phiếu: $\beta$ \al{không là tác động nhân quả}
\end{itemize}
\end{frame}
