\begin{frame}
\frametitle{Khung phân tích: từ đo chất lượng đến hàm ý cho Hà Nội}
\begin{enumerate}
\item \textsb{Sàn \then chất lượng thị trường:} quy tắc giao dịch và mức tập trung lệnh của sàn quyết định chi phí giao dịch và tốc độ giá phản ánh thông tin (\HH{2}, \HH{3})
\item \textsb{Chất lượng \then khả năng đầu tư:} giao dịch đắt và kém thanh khoản làm nhà đầu tư đa dạng hóa nhận lợi suất tương đương chắc chắn thấp hơn (\HH{4})
\item \textsb{Thị trường cổ phiếu \then vị thế trung tâm tài chính:} chỉ là liên hệ có điều kiện, không kiểm định nhân quả (\HH{1})
\item \textsb{Hàm ý:} Bắc Kinh lớn mạnh dù không có sàn cổ phiếu tầm quốc gia (Naeem, 2023) \so Hà Nội chọn chuyên môn hóa: trái phiếu, phái sinh, UPCoM, dữ liệu
\end{enumerate}
\end{frame}

\begin{frame}
\frametitle{Khác biệt kép: so sánh mã chuyển sàn với mã HOSE đối sánh}
\[
A_{it} = \alpha_i + \tau_t + \delta D_i\Post_t + \varepsilon_{it}
\]
\begin{assumption}[Xu hướng song song]
Nếu không chuyển sàn, chênh lệch log Amihud bình quân giữa hai nhóm không đổi theo thời gian:
\[
E[\varepsilon_{it} \mid D_i, \Post_t, \alpha_i, \tau_t] = 0
\]
\end{assumption}
\begin{itemize}
\item $A_{it}$: log Amihud; $D_i = 1$: mã chuyển sàn; $\Post_t = 1$ từ 1/2027; \HH{3}: $\delta < 0$
\item $\alpha_i$ loại khác biệt sẵn có giữa hai nhóm; $\tau_t$ loại thay đổi chung của thị trường (lãi suất, dòng vốn ngoại, nâng hạng)
\item Cửa sổ trước: sau điểm gãy cấu trúc cuối (Bai \& Perron, 1998); báo cáo kèm 7/2023--6/2025
\end{itemize}
\end{frame}

\textsb{Cách làm}
\begin{itemize}

\item Thống kê: $\bar{d}_\text{sau} - \bar{d}_\text{trước}$
\item Kích thước kiểm định kiểm tra bằng mô phỏng trước ngày đăng ký
\end{itemize}
\column{0.48\textwidth}
\textsb{Ý tưởng:} giá trị thật có hiếm không, so với các chuỗi giả không có sự kiện
\begin{enumerate}
\item Ước lượng mô hình AR của $d_t$ trên giai đoạn tham chiếu, hiệu chỉnh độ chệch
\item Tạo chuỗi giả theo mô hình AR, rút phần dư có hoàn lại
\item Tính thống kê trên mỗi chuỗi giả
\item Giá trị p một phía: tỷ lệ chuỗi giả có thống kê nhỏ hơn hoặc bằng giá trị thật
\end{enumerate}
\end{columns}
